<Code example>

<p>Your need to implement function toReadable that converts given number, to readable string.</p>

<p>For example:</p>

toReadable(1); // will return 'one'
toReadable(997); // will return 'nine hundred ninety seven'

<p>Solution</p>

function toReadable (number) {
    const textNumbers = ['zero', 'one', 'two', 'three', 'four', 'five',
        'six', 'seven', 'eight', 'nine', 'ten', 'eleven', 'twelve',
        'thirteen', 'fourteen', 'fifteen', 'sixteen', 'seventeen',
        'eighteen', 'nineteen'];
    const decNumbers = ['twenty', 'thirty', 'forty', 'fifty', 'sixty',
        'seventy', 'eighty', 'ninety'];
    if (number < 20) {
        return textNumbers[number];
    } else if (number < 100) {
        return `${decNumbers[Math.floor(number / 10) - 2]}
            ${toReadable(number % 10)}`.replace(' zero', '' );
    } else {
        return `${textNumbers[Math.floor(number / 100)]} hundred
            ${toReadable(number % 100)}`.replace(' zero', '' );
    }
}
